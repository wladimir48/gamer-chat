const express = require("express"); const app = express(); const http = require("http").Server(app); const io = require("socket.io")(http); const { ExpressPeerServer } = require("peer"); // Раздаем чат и голос из одного места app.use("/peerjs", ExpressPeerServer(http, { debug: true, path: "/" })); app.get("/", (req, res) => { res.send(` GamerChat PRO — FINAL

GamerChat PRO

`); }); let users = {}; io.on("connection", (socket) => { socket.on("join", d => { users[socket.id] = d; io.emit("update", users); }); socket.on("msg", d => io.emit("msg", d)); socket.on("speak", v => io.emit("speak", { id: socket.id, v })); socket.on("disconnect", () => { delete users[socket.id]; io.emit("update", users); }); }); http.listen(3000);
